cmake_minimum_required(VERSION 3.16)
project(cscan VERSION 1.2.2 DESCRIPTION "Native Android pointer-chain scanner" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
file(GLOB ZSTD_SRC CONFIGURE_DEPENDS vendor/zstd/lib/common/*.c vendor/zstd/lib/compress/*.c vendor/zstd/lib/decompress/*.c)
add_library(cscan_zstd STATIC ${ZSTD_SRC})
target_include_directories(cscan_zstd PUBLIC vendor/zstd/lib PRIVATE vendor/zstd/lib/common)
target_compile_definitions(cscan_zstd PRIVATE ZSTD_DISABLE_ASM)
set_target_properties(cscan_zstd PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ASSETS "${CMAKE_CURRENT_BINARY_DIR}/web_assets.hpp")
add_custom_command(OUTPUT "${ASSETS}" COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed.py" "${CMAKE_CURRENT_SOURCE_DIR}/web" "${ASSETS}" DEPENDS web/index.html web/style.css web/app.js scripts/embed.py VERBATIM)
add_library(cscan_core src/memory.cpp src/process.cpp src/readers.cpp src/scan.cpp src/graph.cpp src/graph_scan.cpp src/store.cpp src/text.cpp src/operations.cpp vendor/ahwbp/paradise.c)
target_include_directories(cscan_core PUBLIC src vendor vendor/ahwbp/include)
target_link_libraries(cscan_core PUBLIC Threads::Threads cscan_zstd)
target_compile_options(cscan_core PRIVATE -Wall -Wextra -Wno-misleading-indentation)
add_executable(cscan src/main.cpp src/server.cpp "${ASSETS}")
target_include_directories(cscan PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(cscan PRIVATE cscan_core)
if(ANDROID)
 target_link_options(cscan PRIVATE -Wl,-z,max-page-size=16384)
else()
 enable_testing()
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE cscan_core)
 add_test(NAME core COMMAND core_tests)
 add_executable(graph_benchmark tests/graph_benchmark.cpp)
 target_link_libraries(graph_benchmark PRIVATE cscan_core)
endif()

add_executable(paradise_mock_tests tests/paradise_mock_test.c)
target_include_directories(paradise_mock_tests PRIVATE vendor/ahwbp/include)
add_executable(process_reader_tests tests/process_reader_tests.cpp)
target_link_libraries(process_reader_tests PRIVATE cscan_core)
if(NOT ANDROID)
 add_test(NAME paradise_mock COMMAND paradise_mock_tests)
 add_test(NAME process_reader COMMAND process_reader_tests)
endif()
